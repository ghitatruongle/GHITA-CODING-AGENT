```typescript
  return ip.split('.').reduce((acc, oct) => (acc << 8) + parseInt(oct, 10), 0) >>> 0;

function inCidr(ip: string, base: string, bits: number): boolean {
```

```typescript
import dns from 'node:dns/promises';
import net from 'node:net';

export async function safeFetch(url: string, init: RequestInit = {}): Promise<Response> {
  const parsed = new URL(url);
  const hostname = parsed.hostname;
  const port = parsed.port || (parsed.protocol === 'https:' ? '443' : '80');

  let ip: string;

    ip = hostname;

  const ipUrl = new URL(url);

  headers.set('Host', hostname);

  return fetch(ipUrl.toString(), { ...init, headers });
}
```

### Step 3: Implement PII stream filter with token windowing

```typescript
const PII_PATTERNS = {
  email: /\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b/g,
  phone: /\b(?:\+?\d{1,3}[-.\s]?)?\(?\d{3}\)?[-.\s]?\d{3,4}[-.\s]?\d{4}\b/g,
  ssn: /\b\d{3}-\d{2}-\d{4}\b/g,
};

const REDACTION = {
  email: '[REDACTED_EMAIL]',
  phone: '[REDACTED_PHONE]',
  ssn: '[REDACTED_SSN]',
};

const WINDOW_SIZE = 32; // tokens
const OVERLAP = 8;

export interface PiiStreamFilter {
  process(chunk: string): Promise<string>;
  flush(): Promise<string>;
  reset(): void;
}

export function createPiiStreamFilter(): PiiStreamFilter {
  let buffer: string[] = [];
  let outputTail = ''; // carry-over from previous flush to avoid double-redact

  function redact(text: string): string {
    let out = text;
    for (const [key, pattern] of Object.entries(PII_PATTERNS)) {
      out = out.replace(pattern, REDACTION[key as keyof typeof REDACTION]);
    }
    return out;
  }

  return {
    async process(chunk: string): Promise<string> {
      buffer.push(chunk);
      if (buffer.length < WINDOW_SIZE) return '';

      const joined = buffer.join('');
      const redacted = redact(joined);

      // Release all but the overlap window for the next iteration
      const releaseCount = buffer.length - OVERLAP;
      const released = buffer.slice(0, releaseCount).join('');
      const releasedRedacted = redact(released);

      buffer = buffer.slice(releaseCount);
      outputTail = redacted.slice(releasedRedacted.length);
      return releasedRedacted;
    },
    async flush(): Promise<string> {
      const tail = redact(buffer.join(''));
      buffer = [];
      return outputTail + tail;
    },
    reset(): void {
      buffer = [];
      outputTail = '';
    },
  };
}
```

```typescript
import crypto from 'node:crypto';

export interface SecretRotatorOptions {
  ttlMs: number;
  length: number;
  store?: Map<string, string>; // injectable for tests
}

export interface RotateResult {
  key: string;
  maskedKey: string;
}

export class SecretRotator {
  constructor(private opts: SecretRotatorOptions) {}

  generateKey(): string {
    return crypto.randomBytes(this.opts.length).toString('base64url');
  }

  maskKey(key: string): string {
    if (key.length <= 8) return '*'.repeat(key.length);
    return key.slice(0, 4) + '*'.repeat(key.length - 8) + key.slice(-4);
  }

  async rotate(id: string): Promise<RotateResult> {
    const key = this.generateKey();
    const maskedKey = this.maskKey(key);
    if (this.opts.store) {
      this.opts.store.set(id, key);
    }
    return { key, maskedKey };
  }
}
```

**Part A — Update the `FallbackManager` wrapper to expose an `AbortSignal`:**

```typescript
// packages/ai-engine/src/gateway/fallbackManager.ts
export interface CallOptions {
  signal?: AbortSignal;
}

export class FallbackManager {
  constructor(
    private opts: { timeoutMs: number; logger?: { warn: (msg: string, err?: unknown) => void } },
  ) {}

  async call<T>(fn: (opts: CallOptions) => Promise<T>, opts: CallOptions = {}): Promise<T> {
    // If caller passed a signal, do not double-time; otherwise create one tied to timeoutMs
    const externalSignal = opts.signal;
    const controller = new AbortController();
    let timeoutId: NodeJS.Timeout | null = null;
    let externalAbortHandler: (() => void) | null = null;

    if (!externalSignal) {
      timeoutId = setTimeout(
        () => controller.abort(new Error('FallbackManager timeout')),
        this.opts.timeoutMs,
      );
    } else if (externalSignal.aborted) {
      throw externalSignal.reason ?? new Error('Caller aborted');
    } else {
      externalAbortHandler = () =>
        controller.abort(externalSignal.reason ?? new Error('Caller aborted'));
      externalSignal.addEventListener('abort', externalAbortHandler);
    }

    try {
      // Forward our controller.signal so the inner fetch can react to abort
      return await fn({ signal: controller.signal });
    } finally {
      if (timeoutId !== null) clearTimeout(timeoutId);
      if (externalAbortHandler && externalSignal) {
        externalSignal.removeEventListener('abort', externalAbortHandler);
      }
    }
  }
}
```

```typescript
export function syncToParent(
  child: { id: string; state: any },
  parent: { id: string; state: any },
  previousVersion: number,
): { diff: any } | null {
  if (!parent.id) return null;
  const baseState = previousVersion === 0 ? {} : parent.state;
  const diff = computeDiff(baseState, child.state);
  if (Object.keys(diff).length === 0) return null;
  return { diff };
}

function computeDiff(a: any, b: any): any {
  // shallow diff for now; extend as needed
  const out: any = {};
  for (const k of new Set([...Object.keys(a), ...Object.keys(b)])) {
    if (a[k] !== b[k]) out[k] = b[k];
  }
  return out;
}
```

### Step 4: Fix `spawner.ts`

Replace the immediate-reject path with a queue:

```typescript
export class SubagentSpawner {
  private active = 0;
  private queue: Array<{
    fn: () => Promise<any>;
    resolve: (v: any) => void;
    reject: (e: any) => void;
  }> = [];

  constructor(private opts: { maxConcurrency: number }) {}

  spawn<T>(fn: () => Promise<T>): Promise<T> {
    return new Promise((resolve, reject) => {
      if (this.active < this.opts.maxConcurrency) {
        this.runOne(fn, resolve, reject);
      } else {
        this.queue.push({ fn, resolve, reject });
      }
    });
  }

  private async runOne<T>(fn: () => Promise<T>, resolve: (v: T) => void, reject: (e: any) => void) {
    this.active++;
    try {
      const result = await fn();
      resolve(result);
    } catch (err) {
      reject(err);
    } finally {
      this.active--;
      const next = this.queue.shift();
      if (next) this.runOne(next.fn, next.resolve, next.reject);
    }
  }
}
```

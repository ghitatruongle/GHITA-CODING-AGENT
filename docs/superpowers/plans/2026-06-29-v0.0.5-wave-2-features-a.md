**Step 1 — Extract coordinate math into a pure helper:**

Add this helper at the top of `ScreenPreview.tsx` (or in a sibling file `coordinateMap.ts` if you prefer separation):

```typescript
// apps/mobile/src/components/coordinateMap.ts
export interface DisplayGeometry {
  containerWidth: number;
  containerHeight: number;
  /** Aspect ratio of the underlying screen content (e.g. 16/9) */
  contentAspectRatio: number;
}

export interface NormalizedCoord {
  x: number; // 0..1, clamped
  y: number; // 0..1, clamped
}

/**
 * Compute the on-screen coordinates for a touch at (locationX, locationY),
 * accounting for letterbox/pillarbox introduced by `resizeMode="contain"`.
 *
 * Returns normalized 0..1 values clamped to the visible image area.
 */
export function mapTouchToContent(
  locationX: number,
  locationY: number,
  geom: DisplayGeometry,
): NormalizedCoord {
  const containerRatio = geom.containerWidth / geom.containerHeight;
  let displayWidth = geom.containerWidth;
  let displayHeight = geom.containerHeight;
  let offsetX = 0;
  let offsetY = 0;

  if (containerRatio > geom.contentAspectRatio) {
    // Pillarbox — width is the limiting dimension
    displayWidth = geom.containerHeight * geom.contentAspectRatio;
    offsetX = (geom.containerWidth - displayWidth) / 2;
  } else {
    // Letterbox — height is the limiting dimension
    displayHeight = geom.containerWidth / geom.contentAspectRatio;
    offsetY = (geom.containerHeight - displayHeight) / 2;
  }

  const relativeX = (locationX - offsetX) / displayWidth;
  const relativeY = (locationY - offsetY) / displayHeight;

  return {
    x: Math.max(0, Math.min(1, relativeX)),
    y: Math.max(0, Math.min(1, relativeY)),
  };
}
```

```typescript
/**
 * Extract the first top-level JSON object from a free-form LLM response.
 *
 * Strategy: find the first '{', then walk forward counting nested braces
 * while respecting strings (with backslash escapes). This handles arbitrary
 * nesting depth and avoids the bug where naive `{`/`}` counting chokes on
 * braces that appear inside string values (e.g., `"key": "value with }"`).
 */
export function extractDebateJson(text: string): unknown | null {
  const start = text.indexOf('{');
  if (start === -1) return null;

  let depth = 0;
  let inString = false;
  let escape = false;
  for (let i = start; i < text.length; i++) {
    const ch = text[i];
    if (escape) {
      escape = false;
      continue;
    }
    if (inString) {
      if (ch === '\\') {
        escape = true;
        continue;
      }
      if (ch === '"') {
        inString = false;
      }
      continue;
    }
    if (ch === '"') {
      inString = true;
      continue;
    }
    if (ch === '{') depth++;
    else if (ch === '}') {
      depth--;
      if (depth === 0) {
        const candidate = text.slice(start, i + 1);
        try {
          return JSON.parse(candidate);
        } catch {
          return null; // malformed JSON
        }
      }
    }
  }
  return null; // unbalanced braces
}
```

```typescript
export interface BudgetState {
  version: 1;
  periodStart: string; // ISO-8601
  periodMs: number; // e.g. 30 * 24 * 60 * 60 * 1000 for monthly
  spent: number;
  cap: number;
}

  periodStart: new Date(0).toISOString(),

 * Load budget state from disk. Returns DEFAULT_BUDGET on missing file or

export async function loadBudget(filePath: string): Promise<BudgetState> {
  try {
    const raw = await readFile(filePath, 'utf-8');
    const parsed = JSON.parse(raw) as Partial<BudgetState>;
    if (
      typeof parsed.version === 'number' &&
      typeof parsed.periodStart === 'string' &&
      typeof parsed.periodMs === 'number' &&
      typeof parsed.spent === 'number' &&
      typeof parsed.cap === 'number'
    ) {
      // Migrate older versions if needed
      return { ...DEFAULT_BUDGET, ...parsed, version: 1 } as BudgetState;
    }
    console.warn(`[budget] Invalid budget schema at ${filePath}; using defaults`);
    return DEFAULT_BUDGET;
  } catch (err: any) {
    if (err.code !== 'ENOENT') {
      console.warn(`[budget] Failed to read ${filePath}: ${err.message}; using defaults`);
    }
    return DEFAULT_BUDGET;
  }
}

/**
 * Persist budget state to disk. Creates the directory if it does not exist.
 */
export async function saveBudget(filePath: string, state: BudgetState): Promise<void> {
  await mkdir(dirname(filePath), { recursive: true });
  await writeFile(filePath, JSON.stringify(state, null, 2), 'utf-8');
}

/**
 * Check if the current period has elapsed; if so, reset `spent` and bump
 * `periodStart`. Returns the (possibly reset) state.
 */
export function rollPeriodIfNeeded(state: BudgetState, now = Date.now()): BudgetState {
  const periodStartMs = Date.parse(state.periodStart);
  if (Number.isNaN(periodStartMs)) {
    return { ...state, periodStart: new Date(now).toISOString(), spent: 0 };
  }
  if (now - periodStartMs >= state.periodMs) {
    return {
      ...state,
      periodStart: new Date(now).toISOString(),
      spent: 0,
    };
  }
  return state;
}

 * Falls back to `~/.ghita/budget.json` in non-Tauri contexts (tests, CLI).
 */
export function defaultBudgetPath(homeDir: string): string {
  return join(homeDir, '.ghita', 'budget.json');
}
```

```typescript
import {
  defaultBudgetPath,
  loadBudget,
  saveBudget,
  rollPeriodIfNeeded,
  type BudgetState,
} from './budget-persistence';

export class Budget {
  private state: BudgetState;
  private filePath: string;

  constructor(opts: { filePath?: string; homeDir?: string; initialState?: BudgetState }) {
    this.filePath =
      opts.filePath ??
      (opts.homeDir
        ? defaultBudgetPath(opts.homeDir)
        : defaultBudgetPath(process.env.HOME ?? '/tmp'));
    this.state = opts.initialState ?? DEFAULT_BUDGET;
  }

  async load(): Promise<void> {
    this.state = await loadBudget(this.filePath);
    this.state = rollPeriodIfNeeded(this.state);
    await this.save();
  }

  async save(): Promise<void> {
    await saveBudget(this.filePath, this.state);
  }

  spent(): number {
    return this.state.spent;
  }
  cap(): number {
    return this.state.cap;
  }
  remaining(): number {
    return Math.max(0, this.state.cap - this.state.spent);
  }

  async addSpent(amount: number): Promise<void> {
    this.state = rollPeriodIfNeeded({ ...this.state, spent: this.state.spent + amount });
    await this.save();
  }

  async setCap(cap: number): Promise<void> {
    if (cap < 0) throw new Error('cap must be non-negative');
    this.state = { ...this.state, cap };
    await this.save();
  }

  snapshot(): BudgetState {

  }
}
```

```rust
async fn read_pty_to_channel(
    mut reader: Box<dyn AsyncRead + Send + Unpin>,
    tx: mpsc::Sender<String>,
    sessions: Arc<DashMap<String, Arc<Mutex<SessionHandle>>>>,
    session_id: String,
) {
    let mut buf = [0u8; 4096];
    let mut carry: Vec<u8> = Vec::with_capacity(8192);

    loop {
        match reader.read(&mut buf).await {
            Ok(0) => break, // EOF
            Ok(n) => {
                carry.extend_from_slice(&buf[..n]);
                loop {
                    match std::str::from_utf8(&carry) {
                        Ok(s) => {
                            if tx.send(s.to_string()).await.is_err() {
                                return; // receiver dropped
                            }
                            carry.clear();
                            break;
                        }
                        Err(e) => {
                            let valid_up_to = e.valid_up_to();
                            if valid_up_to > 0 {
                                let s = std::str::from_utf8(&carry[..valid_up_to])
                                    .unwrap_or("");
                                if tx.send(s.to_string()).await.is_err() {
                                    return;
                                }
                                carry.drain(..valid_up_to);
                            } else {
                                // No complete UTF-8 yet; wait for more bytes
                                break;
                            }
                        }
                    }
                }
            }
            Err(e) => {
                eprintln!("PTY read error: {}", e);
                break;
            }
        }
    }

    // EOF cleanup — remove session from map
    sessions.remove(&session_id);
```

```typescript
import { create } from 'zustand';
import { useEffect } from 'react';
import { invoke } from '@tauri-apps/api/tauri';
import { NotificationHistory, type Notification } from '@ghita/notification';

interface NotificationStore {
  items: Notification[];
  history: NotificationHistory;
  refresh: () => void;
  markAllRead: () => void;
  push: (n: Omit<Notification, 'id' | 'createdAt' | 'read'>) => Promise<void>;
}

export const useNotificationStore = create<NotificationStore>((set, get) => ({
  items: [],
  history: new NotificationHistory(),

  refresh: () => {
    const items = get().history.list();
    set({ items });
  },

  markAllRead: () => {
    get().history.markAllRead();
    get().refresh();
  },

  push: async (n) => {
    const note: Notification = {
      id: crypto.randomUUID(),
      createdAt: new Date().toISOString(),
      read: false,
      ...n,
    };
    get().history.add(note);
    // Also show OS-level notification
    try {
      await invoke('show_notification', { title: note.title, body: note.body });
    } catch (e) {
      console.warn('OS notification failed (may be sandboxed):', e);
    }
    get().refresh();
  },
}));

/** Convenience hook for components that just need to read */
export function useNotifications() {
  const items = useNotificationStore((s) => s.items);
  const unreadCount = items.filter((n) => !n.read).length;
  const markAllRead = useNotificationStore((s) => s.markAllRead);
  const refresh = useNotificationStore((s) => s.refresh);

  useEffect(() => {
    refresh();
  }, [refresh]);

  return { items, unreadCount, markAllRead, refresh };
}
```

```tsx
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
```

```typescript
// apps/desktop/src/hooks/useMonitoring.ts
import { create } from 'zustand';
import {
  ErrorGrouper,
  AlertEngine,
  UsageTelemetry,
  type GroupedError,
  type Alert,
  type TelemetrySnapshot,
} from '@ghita/monitoring';

interface MonitoringStore {
  errors: GroupedError[];
  alerts: Alert[];
  telemetry: TelemetrySnapshot | null;
  grouper: ErrorGrouper;
  alertEngine: AlertEngine;
  usage: UsageTelemetry;
  refresh: () => void;
  acknowledgeAlert: (id: string) => void;
}

export const useMonitoringStore = create<MonitoringStore>((set, get) => {
  const grouper = new ErrorGrouper();
  const alertEngine = new AlertEngine();
  const usage = new UsageTelemetry();

  return {
    errors: [],
    alerts: [],
    telemetry: null,
    grouper,
    alertEngine,
    usage,

    refresh: () => {
      const errors = grouper.list();
      const alerts = alertEngine.active();
      const telemetry = usage.snapshot();
      set({ errors, alerts, telemetry });
    },

    acknowledgeAlert: (id) => {
      get().alertEngine.acknowledge(id);
      get().refresh();
    },
  };
});

export function useMonitoring(autoRefreshMs = 30_000) {
  const store = useMonitoringStore();
  // Auto-refresh
  if (typeof window !== 'undefined') {
    const refresh = store.refresh;
    setInterval(refresh, autoRefreshMs);
  }
  return store;
}
```

```tsx
// apps/desktop/src/components/ErrorRateChart.tsx
import {
  LineChart,
  Line,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  ResponsiveContainer,
  Legend,
} from 'recharts';
import type { GroupedError } from '@ghita/monitoring';

export function ErrorRateChart({ errors }: { errors: GroupedError[] }) {
  // Group errors into time buckets (last hour, bucketed by minute)
  const now = Date.now();
  const data: Array<{ time: string; count: number }> = [];
  for (let i = 59; i >= 0; i--) {
    const bucketStart = now - i * 60_000;
    const label = new Date(bucketStart).toLocaleTimeString([], {
      hour: '2-digit',
      minute: '2-digit',
    });
    const count = errors.filter((e) => {
      const t = Date.parse(e.lastSeen);
      return t >= bucketStart && t < bucketStart + 60_000;
    }).length;
    data.push({ time: label, count });
  }

  if (data.every((d) => d.count === 0)) {
    return (
      <div className="empty-chart" role="status">
        No errors recorded in the last hour.
      </div>
    );
  }

  return (
    <ResponsiveContainer width="100%" height={240}>
      <LineChart data={data}>
        <CartesianGrid strokeDasharray="3 3" stroke="var(--border-color)" />
        <XAxis dataKey="time" stroke="var(--text-secondary)" />
        <YAxis stroke="var(--text-secondary)" allowDecimals={false} />
        <Tooltip
          contentStyle={{ background: 'var(--panel-bg)', border: '1px solid var(--border-color)' }}
        />
        <Legend />
        <Line type="monotone" dataKey="count" stroke="var(--accent-primary)" name="Errors" />
      </LineChart>
    </ResponsiveContainer>
  );
}
```

**Step 1 — Rust commands for budget persistence:**

```rust
// apps/desktop/src-tauri/src/lib.rs

use serde::{Serialize, Deserialize};

#[derive(Serialize, Deserialize, Clone, Debug)]
#[serde(rename_all = "camelCase")]
pub struct BudgetState {
    pub version: u32,
    pub period_start: String,
    pub period_ms: u64,
    pub spent: f64,
    pub cap: f64,
}

impl Default for BudgetState {
    fn default() -> Self {
        Self {
            version: 1,
            period_start: chrono::Utc::now().to_rfc3339(),
            period_ms: 30 * 24 * 60 * 60 * 1000, // 30 days
            spent: 0.0,
            cap: 100.0,
        }
    }
}

fn budget_path(app: &tauri::AppHandle) -> Result<std::path::PathBuf, String> {
    let dir = app.path().app_data_dir().map_err(|e| e.to_string())?;
    std::fs::create_dir_all(&dir).map_err(|e| e.to_string())?;
    Ok(dir.join("budget.json"))
}

#[tauri::command]
fn read_budget(app: tauri::AppHandle) -> Result<BudgetState, String> {
    let path = budget_path(&app)?;
    match std::fs::read_to_string(&path) {
        Ok(s) => serde_json::from_str(&s).map_err(|e| e.to_string()),
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(BudgetState::default()),
        Err(e) => Err(e.to_string()),
    }
}

#[tauri::command]
fn write_budget(app: tauri::AppHandle, state: BudgetState) -> Result<(), String> {
    let path = budget_path(&app)?;
    let tmp = path.with_extension("json.tmp");
    std::fs::write(&tmp, serde_json::to_string_pretty(&state).map_err(|e| e.to_string())?)
        .map_err(|e| e.to_string())?;
    std::fs::rename(&tmp, &path).map_err(|e| e.to_string())?;
    Ok(())
}
```

```typescript
// apps/desktop/src/hooks/useBudget.ts
import { useState, useEffect, useCallback } from 'react';
import { invoke } from '@tauri-apps/api/tauri';

export interface BudgetState {
  version: number;
  periodStart: string;
  periodMs: number;
  spent: number;
  cap: number;
}

export function useBudget() {
  const [state, setState] = useState<BudgetState | null>(null);
  const [error, setError] = useState<string | null>(null);

  const refresh = useCallback(async () => {
    try {
      const s = await invoke<BudgetState>('read_budget');
      setState(s);
    } catch (e) {
      setError(String(e));
    }
  }, []);

  const update = useCallback(
    async (patch: Partial<BudgetState>) => {
      if (!state) return;
      const next = { ...state, ...patch, version: 1 };
      try {
        await invoke('write_budget', { state: next });
        setState(next);
      } catch (e) {
        setError(String(e));
      }
    },
    [state],
  );

  useEffect(() => {
    refresh();
  }, [refresh]);

  return { state, error, refresh, update };
}
```

```tsx
export function QuotaGauge({ spent, cap }: { spent: number; cap: number }) {
  const { t } = useTranslation();
```

```tsx
// apps/desktop/src/components/ResetSchedule.tsx
import { useTranslation } from 'react-i18next';

export function ResetSchedule({
  periodStart,
  periodMs,
}: {
  periodStart: string;
  periodMs: number;
}) {
  const { t } = useTranslation();
  const start = new Date(periodStart);
  const nextReset = new Date(start.getTime() + periodMs);
  const now = Date.now();
  const msUntilReset = nextReset.getTime() - now;

  function formatDuration(ms: number): string {
    if (ms <= 0) return t('quota.resetImminent');
    const days = Math.floor(ms / 86_400_000);
    const hours = Math.floor((ms % 86_400_000) / 3_600_000);
    return t('quota.resetsIn', { days, hours });
  }

  return (
    <div className="reset-schedule">
      <h3>{t('quota.resetSchedule')}</h3>
      <p>
        <strong>{t('quota.currentPeriod')}</strong>: {start.toLocaleString()}
      </p>
      <p>
        <strong>{t('quota.nextReset')}</strong>: {nextReset.toLocaleString()} (
        {formatDuration(msUntilReset)})
      </p>
    </div>
  );
}
```

```tsx
        <UsageTable />
      </section>

      </section>

          id="quota-cap-input"
          type="number"
          min={0}
          step={1}
          defaultValue={state.cap}

            }
          }}
```

```rust
// apps/desktop/src-tauri/src/lib.rs

use std::sync::atomic::{AtomicBool, Ordering};
use std::sync::Arc;

#[derive(serde::Serialize, serde::Deserialize, Clone)]
pub struct KnowledgeGraphDto {
    pub symbols: Vec<SymbolDto>,
    pub relations: Vec<RelationDto>,
}

#[derive(serde::Serialize, serde::Deserialize, Clone)]
pub struct SymbolDto {
    pub id: String,
    pub name: String,
    pub kind: String, // "function" | "class" | "variable" | "import"
    pub file: String,
    pub line: u32,
}

#[derive(serde::Serialize, serde::Deserialize, Clone)]
pub struct RelationDto {
    pub from: String,
    pub to: String,
    pub kind: String, // "calls" | "imports" | "extends" | "uses"
}

// Use a global atomic for cancellation
static PARSE_CANCELLED: AtomicBool = AtomicBool::new(false);

#[tauri::command]
async fn parse_workspace(path: String) -> Result<KnowledgeGraphDto, String> {
    use ghita_code_graph::KnowledgeGraph;
    PARSE_CANCELLED.store(false, Ordering::SeqCst);
    let cancelled = Arc::new(&PARSE_CANCELLED); // can't capture static in async; use unsafe or thread-local

    // For simplicity, build synchronously and check flag
    let graph = tokio::task::spawn_blocking(move || {
        KnowledgeGraph::build_from_path_with_cancel(&path, || cancelled.load(Ordering::SeqCst))
    })
    .await
    .map_err(|e| e.to_string())?
    .map_err(|e| e.to_string())?;

    Ok(KnowledgeGraphDto {
        symbols: graph.symbols().iter().map(|s| SymbolDto {
            id: s.id.clone(),
            name: s.name.clone(),
            kind: s.kind.to_string(),
            file: s.file.clone(),
            line: s.line,
        }).collect(),
        relations: graph.relations().iter().map(|r| RelationDto {
            from: r.from.clone(),
            to: r.to.clone(),
            kind: r.kind.to_string(),
        }).collect(),
    })
}

#[tauri::command]
fn cancel_parse() {
    PARSE_CANCELLED.store(true, Ordering::SeqCst);
}
```

```typescript
// apps/desktop/src/hooks/useCodeGraph.ts
import { useState, useCallback } from 'react';
import { invoke } from '@tauri-apps/api/tauri';

export interface Symbol {
  id: string;
  name: string;
  kind: 'function' | 'class' | 'variable' | 'import';
  file: string;
  line: number;
}

export interface Relation {
  from: string;
  to: string;
  kind: 'calls' | 'imports' | 'extends' | 'uses';
}

export interface KnowledgeGraph {
  symbols: Symbol[];
  relations: Relation[];
}

export function useCodeGraph() {
  const [graph, setGraph] = useState<KnowledgeGraph | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const build = useCallback(async (workspacePath: string) => {
    setLoading(true);
    setError(null);
    try {
      const g = await invoke<KnowledgeGraph>('parse_workspace', { path: workspacePath });
      setGraph(g);
    } catch (e) {
      setError(String(e));
      setGraph(null);
    } finally {
      setLoading(false);
    }
  }, []);

  const cancel = useCallback(async () => {
    try {
      await invoke('cancel_parse');
    } catch {
      // best-effort
    }
  }, []);

  return { graph, loading, error, build, cancel };
}
```

**Step 5 — `CodeGraphView.tsx`:**

```tsx
// apps/desktop/src/views/CodeGraphView.tsx
import { useState } from 'react';
import { useTranslation } from 'react-i18next';
import { useCodeGraph } from '../hooks/useCodeGraph';
import { CodeGraph } from '../components/CodeGraph';
import { SymbolSearch } from '../components/SymbolSearch';

export function CodeGraphView() {
  const { t } = useTranslation();
  const { graph, loading, error, build, cancel } = useCodeGraph();
  const [workspacePath, setWorkspacePath] = useState('');
  const [filter, setFilter] = useState('');

  async function handleBuild() {
    if (!workspacePath) return;
    await build(workspacePath);
  }

  const filtered =
    graph && filter
      ? {
          ...graph,
          symbols: graph.symbols.filter((s) => s.name.toLowerCase().includes(filter.toLowerCase())),
          relations: graph.relations.filter((r) => {
            const fromMatch = graph.symbols.some(
              (s) => s.id === r.from && s.name.toLowerCase().includes(filter.toLowerCase()),
            );
            const toMatch = graph.symbols.some(
              (s) => s.id === r.to && s.name.toLowerCase().includes(filter.toLowerCase()),
            );
            return fromMatch || toMatch;
          }),
        }
      : graph;

  return (
    <main className="code-graph-view" aria-labelledby="codegraph-title">
      <header className="code-graph-view-header">
        <h1 id="codegraph-title">{t('codegraph.title')}</h1>
      </header>

      <div className="code-graph-controls">
        <label htmlFor="codegraph-path">{t('codegraph.workspacePath')}</label>
        <input
          id="codegraph-path"
          type="text"
          value={workspacePath}
          onChange={(e) => setWorkspacePath(e.target.value)}
          placeholder="/path/to/workspace"
          className="focus-ring"
        />
        {loading ? (
          <button type="button" className="focus-ring" onClick={cancel}>
            {t('common.cancel')}
          </button>
        ) : (
          <button
            type="button"
            className="focus-ring"
            onClick={handleBuild}
            disabled={!workspacePath}
          >
            {t('codegraph.build')}
          </button>
        )}
      </div>

      {error && (
        <p className="code-graph-error" role="alert">
          {t('codegraph.buildError', { error })}
        </p>
      )}

      {loading && <p className="code-graph-loading">{t('codegraph.building')}</p>}

      {graph && (
        <>
          <SymbolSearch value={filter} onChange={setFilter} />
          <div className="code-graph-stats" aria-live="polite">
            {t('codegraph.stats', {
              symbols: filtered?.symbols.length ?? 0,
              relations: filtered?.relations.length ?? 0,
            })}
          </div>
          {filtered && <CodeGraph graph={filtered} />}
        </>
      )}

      {!graph && !loading && !error && <p className="code-graph-empty">{t('codegraph.empty')}</p>}
    </main>
  );
}
```

```tsx
// apps/desktop/src/components/CodeGraph.tsx
import { useMemo, useCallback } from 'react';
import ReactFlow, {
  Node,
  Edge,
  Controls,
  MiniMap,
  Background,
  BackgroundVariant,
  useNodesState,
  useEdgesState,
  addEdge,
  Connection,
} from 'reactflow';
import 'reactflow/dist/style.css';
import type { KnowledgeGraph } from '../hooks/useCodeGraph';

const KIND_COLORS: Record<string, string> = {
  function: '#6366f1',
  class: '#ec4899',
  variable: '#10b981',
  import: '#f59e0b',
};

export function CodeGraph({ graph }: { graph: KnowledgeGraph }) {
  const initialNodes: Node[] = useMemo(
    () =>
      graph.symbols.map((s, i) => ({
        id: s.id,
        position: { x: (i % 10) * 200, y: Math.floor(i / 10) * 120 },
        data: { label: `${s.name}\n(${s.kind})` },
        style: {
          background: KIND_COLORS[s.kind] ?? '#888',
          color: 'white',
          border: 'none',
          borderRadius: 4,
          padding: 8,
          fontSize: 11,
          whiteSpace: 'pre' as const,
        },
      })),
    [graph.symbols],
  );

  const initialEdges: Edge[] = useMemo(
    () =>
      graph.relations.map((r) => ({
        id: `${r.from}-${r.to}`,
        source: r.from,
        target: r.to,
        label: r.kind,
        animated: r.kind === 'calls',
      })),
    [graph.relations],
  );

  const [nodes, , onNodesChange] = useNodesState(initialNodes);
  const [edges, setEdges, onEdgesChange] = useEdgesState(initialEdges);

  const onConnect = useCallback(
    (params: Connection) => setEdges((eds) => addEdge({ ...params, animated: true }, eds)),
    [setEdges],
  );

  return (
    <div className="code-graph-canvas">
      <ReactFlow
        nodes={nodes}
        edges={edges}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        onConnect={onConnect}
        fitView
      >
        <Controls />
        <MiniMap />
        <Background variant={BackgroundVariant.Dots} />
      </ReactFlow>
    </div>
  );
}
```

**Step 1 — Add cache to the package:**

```typescript
// packages/code-graph/src/cache.ts
import { readFile, writeFile, stat } from 'node:fs/promises';
import { join } from 'node:path';
import type { KnowledgeGraph } from './types';

export interface CacheEntry {
  workspacePath: string;
  builtAt: string;
  fileMtimes: Record<string, number>;
  graph: KnowledgeGraph;
}

export async function loadCache(
  cachePath: string,
  workspacePath: string,
): Promise<KnowledgeGraph | null> {
  try {
    const raw = await readFile(cachePath, 'utf-8');
    const entry: CacheEntry = JSON.parse(raw);
    if (entry.workspacePath !== workspacePath) return null;
    // Verify all mtimes match
    for (const [file, expectedMtime] of Object.entries(entry.fileMtimes)) {
      try {
        const s = await stat(join(workspacePath, file));
        if (Math.floor(s.mtimeMs) !== expectedMtime) return null;
      } catch {
        return null;
      }
    }
    return entry.graph;
  } catch {
    return null;
  }
}

export async function saveCache(
  cachePath: string,
  workspacePath: string,
  fileMtimes: Record<string, number>,
  graph: KnowledgeGraph,
): Promise<void> {
  const entry: CacheEntry = {
    workspacePath,
    builtAt: new Date().toISOString(),
    fileMtimes,
    graph,
  };
  await writeFile(cachePath, JSON.stringify(entry), 'utf-8');
}
```

```typescript
// apps/desktop/src/hooks/useVoiceInput.ts
import { useEffect, useRef, useState, useCallback } from 'react';

interface SpeechRecognitionResultEvent extends Event {
  results: SpeechRecognitionResultList;
  resultIndex: number;
}

interface SpeechRecognitionErrorEvent extends Event {
  error: string;
  message?: string;
}

interface SpeechRecognitionLike extends EventTarget {
  continuous: boolean;
  interimResults: boolean;
  lang: string;
  start(): void;
  stop(): void;
  abort(): void;
  onresult: ((e: SpeechRecognitionResultEvent) => void) | null;
  onerror: ((e: SpeechRecognitionErrorEvent) => void) | null;
  onend: (() => void) | null;
  onstart: (() => void) | null;
}

declare global {
  interface Window {
    SpeechRecognition?: { new (): SpeechRecognitionLike };
    webkitSpeechRecognition?: { new (): SpeechRecognitionLike };
  }
}

export type VoiceSupport = 'supported' | 'webkit-only' | 'unsupported';

export function detectVoiceSupport(): VoiceSupport {
  if (typeof window === 'undefined') return 'unsupported';
  if (window.SpeechRecognition) return 'supported';
  if (window.webkitSpeechRecognition) return 'webkit-only';
  return 'unsupported';
}

export interface VoiceInput {
  listening: boolean;
  transcript: string;
  error: string | null;
  support: VoiceSupport;
  start: () => void;
  stop: () => void;
  reset: () => void;
}

export function useVoiceInput(lang = 'en-US'): VoiceInput {
  const support = useRef(detectVoiceSupport()).current;
  const [listening, setListening] = useState(false);
  const [transcript, setTranscript] = useState('');
  const [error, setError] = useState<string | null>(null);
  const recognitionRef = useRef<SpeechRecognitionLike | null>(null);

  useEffect(() => {
    if (support === 'unsupported') {
      setError('Speech recognition not supported in this environment');
      return;
    }
    const Ctor = window.SpeechRecognition ?? window.webkitSpeechRecognition!;
    const rec = new Ctor();
    rec.continuous = false;
    rec.interimResults = true;
    rec.lang = lang;

    rec.onresult = (e) => {
      const text = Array.from(e.results)
        .map((r) => r[0].transcript)
        .join(' ');
      setTranscript(text);
    };

    rec.onerror = (e) => {
      setError(`${e.error}${e.message ? `: ${e.message}` : ''}`);
      setListening(false);
    };

    rec.onend = () => {
      setListening(false);
    };

    recognitionRef.current = rec;

    return () => {
      try {
        rec.abort();
      } catch {
        /* noop */
      }
    };
  }, [lang, support]);

  const start = useCallback(() => {
    if (!recognitionRef.current) return;
    setError(null);
    setTranscript('');
    try {
      recognitionRef.current.start();
      setListening(true);
    } catch (e) {
      setError(String(e));
    }
  }, []);

  const stop = useCallback(() => {
    if (!recognitionRef.current) return;
    try {
      recognitionRef.current.stop();
    } catch {
      setListening(false);
    }
  }, []);

  const reset = useCallback(() => {
    setTranscript('');
    setError(null);
  }, []);

  return { listening, transcript, error, support, start, stop, reset };
}
```

```tsx
// apps/desktop/src/components/VoiceWaveform.tsx
import { useEffect, useRef, useState } from 'react';
import { useTranslation } from 'react-i18next';
import { AudioRingBuffer } from '@ghita/voice';

interface VoiceWaveformProps {
  active: boolean;
  barCount?: number;
  maxBarHeight?: number;
}

export function VoiceWaveform({ active, barCount = 16, maxBarHeight = 40 }: VoiceWaveformProps) {
  const { t } = useTranslation();
  const [amplitudes, setAmplitudes] = useState<number[]>(() => new Array(barCount).fill(0));
  const bufferRef = useRef<AudioRingBuffer | null>(null);
  const rafRef = useRef<number | null>(null);

  useEffect(() => {
    if (!active) {
      setAmplitudes(new Array(barCount).fill(0));
      return;
    }

    let stream: MediaStream | null = null;
    let audioContext: AudioContext | null = null;
    let analyser: AnalyserNode | null = null;

    async function setup() {
      try {
        stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        audioContext = new AudioContext();
        const source = audioContext.createMediaStreamSource(stream);
        analyser = audioContext.createAnalyser();
        analyser.fftSize = 64;
        source.connect(analyser);
        bufferRef.current = new AudioRingBuffer(barCount);

        const data = new Uint8Array(analyser.frequencyBinCount);
        const tick = () => {
          if (!analyser) return;
          analyser.getByteFrequencyData(data);
          // Average amplitude across bins
          const avg = data.reduce((s, v) => s + v, 0) / data.length / 255;
          bufferRef.current!.push(avg);
          setAmplitudes(bufferRef.current!.snapshot());
          rafRef.current = requestAnimationFrame(tick);
        };
        tick();
      } catch (e) {
        console.warn('Microphone access denied:', e);
      }
    }

    setup();

    return () => {
      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
      stream?.getTracks().forEach((t) => t.stop());
      audioContext?.close();
      analyser = null;
    };
  }, [active, barCount]);

  return (
    <div className="voice-waveform" aria-label={t('voice.waveform')} aria-hidden="true">
      {amplitudes.map((a, i) => (
        <span
          key={i}
          className="voice-waveform-bar"
          style={{ height: `${Math.max(2, a * maxBarHeight)}px` }}
        />
      ))}
    </div>
  );
}
```

```typescript
export function useTTS(provider?: TtsProvider) {
  const providerRef = useRef<TtsProvider>(provider ?? new SilentTtsProvider());
  const [speaking, setSpeaking] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const speak = useCallback(async (text: string, opts?: { voice?: string; rate?: number }) => {
    try {
      setError(null);
      setSpeaking(true);
      await providerRef.current.speak(text, opts);
    } catch (e) {
      setError(String(e));
    } finally {
      setSpeaking(false);
    }
  }, []);

  const stop = useCallback(async () => {
    try {
      await providerRef.current.stop();
    } finally {
      setSpeaking(false);
    }
  }, []);

  useEffect(() => {
    return () => {
      providerRef.current.stop().catch(() => {});
    };
  }, []);

  return { speak, stop, speaking, error };
}
```

```javascript
#!/usr/bin/env node
import { spawnSync } from 'node:child_process';
import { existsSync } from 'node:fs';
import { resolve } from 'node:path';
import process from 'node:process';

const IOS_DIR = resolve(process.cwd(), 'apps/mobile/ios');
const WORKSPACE = resolve(IOS_DIR, 'GhitaMobile.xcworkspace');
const SCHEME = 'GhitaMobile';
const CONFIGURATION = process.env.IOS_CONFIGURATION ?? 'Debug';
const DESTINATION = process.env.IOS_DESTINATION ?? 'generic/platform=iOS Simulator';

function run(cmd, args, opts = {}) {
  console.log(`> ${cmd} ${args.join(' ')}`);
  const result = spawnSync(cmd, args, { stdio: 'inherit', ...opts });
  if (result.status !== 0) {
    throw new Error(`${cmd} exited with code ${result.status}`);
  }
}

if (!existsSync(WORKSPACE)) {
  console.error(`Workspace not found at ${WORKSPACE}`);
  console.error('Run `pnpm --filter @ghita/mobile pod-install` first.');
  process.exit(1);
}

console.log('[ios-sign] Installing CocoaPods...');
run('pod', ['install', '--project-directory', IOS_DIR]);

console.log('[ios-sign] Building iOS app (CODE_SIGNING_ALLOWED=NO)...');
run('xcodebuild', [
  '-workspace', WORKSPACE,
  '-scheme', SCHEME,
  '-configuration', CONFIGURATION,
  '-destination', DESTINATION,
  '-derivedDataPath', resolve(IOS_DIR, 'build'),
  'CODE_SIGNING_ALLOWED=NO',
  'CODE_SIGNING_REQUIRED=NO',
  'CODE_SIGN_IDENTITY=',
  'build',
]);

console.log('[ios-sign] Build complete.');
console.log(`[ios-sign] App: ${resolve(IOS_DIR, 'build/Build/Products/${CONFIGURATION === 'Debug' ? 'Debug-iphonesimulator' : 'Release-iphonesimulator'}/GhitaMobile.app')}`);
```
